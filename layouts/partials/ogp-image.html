{{- /* ogp-image.html — builds a per-page OGP (Twitter Card) image by overlaying the page title onto the fixed themed background (assets/ogp/base.png) with the images.Text filter, and returns its absolute URL. images.Text only wraps on whitespace, which never occurs in Japanese titles, so we wrap manually. The title is tokenised into ASCII alphanumeric "words" (kept intact) and single other runes, then greedily packed into lines: each non-ASCII rune counts as 2 width units and each ASCII character as 1, breaking near maxUnits and capping the height at maxLines (with an ellipsis when truncated). Keeping ASCII words intact avoids breaking words like "PlantUML" mid-word. Coordinates are aligned with the background drawn by cmd/ogp: the title sits below the accent bar and above the bottom-left branding. When params.ogp.siteName is set the site name is drawn in that bottom-left branding area as a second overlay — set it to true to use .Site.Title, or to a string for a custom label. It is opt-in, so sites that don't enable it keep their existing generated images unchanged. Returns an empty string when inputs are missing or the page has no title, so callers can fall back to other images. */ -}} {{- $url := "" -}} {{- if and .IsPage .Title -}} {{- $base := resources.Get "ogp/base.png" -}} {{- $font := resources.Get "ogp/NotoSansJP-Bold.ttf" -}} {{- if and $base $font -}} {{- $maxUnits := 28 -}} {{- $maxLines := 3 -}} {{- $lines := slice -}} {{- $line := "" -}} {{- $w := 0 -}} {{- range $tok := (findRE `[A-Za-z0-9]+|.` .Title) -}} {{- /* An ASCII word has byte length == rune count; any other token is a single (usually full-width) rune counted as 2 units. */ -}} {{- $ascii := eq (len $tok) (len (split $tok "")) -}} {{- $tw := cond $ascii (len $tok) 2 -}} {{- $isSpace := in (slice " " "　") $tok -}} {{- if and (gt (add $w $tw) $maxUnits) (gt (len $line) 0) -}} {{- $lines = $lines | append (trim $line " 　") -}} {{- $line = "" -}} {{- $w = 0 -}} {{- end -}} {{- if not (and $isSpace (eq (len $line) 0)) -}} {{- $line = printf "%s%s" $line $tok -}} {{- $w = add $w $tw -}} {{- end -}} {{- end -}} {{- with (trim $line " 　") }}{{ $lines = $lines | append . }}{{- end -}} {{- /* first caps safely when len <= maxLines, so this covers both cases; the ellipsis is only appended when lines were actually dropped. */ -}} {{- $text := delimit (first $maxLines $lines) "\n" -}} {{- if gt (len $lines) $maxLines }}{{ $text = printf "%s…" $text }}{{ end -}} {{- /* x = cmd/ogp margin+pad (72+48); y sits just below the accent bar. */ -}} {{- $opts := dict "color" "#ffffff" "size" 60 "x" 120 "y" 150 "linespacing" 14 "font" $font -}} {{- $filters := slice (images.Text $text $opts) -}} {{- /* Optional site-name branding, bottom-left, aligned with cmd/ogp (margin+pad = 120; branding band near the panel's lower-left). Opt-in via params.ogp.siteName: true uses .Site.Title, a string draws itself. */ -}} {{- $siteName := "" -}} {{- with .Site.Params.ogp -}} {{- with .siteName -}} {{- $siteName = cond (eq (printf "%T" .) "bool") $.Site.Title (printf "%v" .) -}} {{- end -}} {{- end -}} {{- with $siteName -}} {{- $brandOpts := dict "color" "#ffffff" "size" 30 "x" 120 "y" 452 "font" $font -}} {{- $filters = $filters | append (images.Text . $brandOpts) -}} {{- end -}} {{- $img := $base.Filter $filters -}} {{- $url = $img.Permalink -}} {{- end -}} {{- end -}} {{- return $url -}}
