{{- /* Render an Open Graph link-preview card for a URL. Falls back to a plain anchor when OGP fetching is disabled or the target exposes no usable metadata, so content always renders something clickable. Input: dict { url, text } text = anchor label used for the fallback link (defaults to url) */ -}} {{- $url := .url -}} {{- $text := or .text $url -}} {{- $external := or (hasPrefix $url "http://") (hasPrefix $url "https://") -}} {{- /* On by default; opt out with params.ogpCard.enabled = false. */ -}} {{- $enabled := ne site.Params.ogpCard.enabled false -}} {{- $ogp := dict "ok" false -}} {{- if and $enabled $external -}} {{- $ogp = partialCached "helper/ogp.html" (dict "url" $url) $url -}} {{- end -}} {{- if $ogp.ok -}} {{- with $ogp.image }} {{- end }} {{ $ogp.title }} {{- with $ogp.description }}{{ . }}{{ end }} {{ partial "helper/icon" "link" }} {{ $ogp.siteName }} {{- else -}} {{ $text | safeHTML }} {{- end -}}
