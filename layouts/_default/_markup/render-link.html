{{- /* Link render hook. A bare external URL (an autolink, where the visible text equals the destination) is rendered as an Open Graph preview card. Every other link — labelled markdown links, relative/internal links, anchors, mailto — renders as a normal anchor, matching Hugo's default output. Set `params.ogpCard.enabled = false` in site config to fall back to plain links everywhere. */ -}} {{- $dest := .Destination -}} {{- $plain := .PlainText | default "" -}} {{- $external := or (hasPrefix $dest "http://") (hasPrefix $dest "https://") -}} {{- $isBareURL := and $external (eq (trim $plain " ") (trim $dest " ")) -}} {{- if $isBareURL -}} {{- /* Mark only the first OGP card on the page as eager. Hugo renders (and memoizes) a page's content once, so the first card is reliably the one nearest the top and the likely LCP element: it is fetched eagerly with a high priority, while every later card stays lazy. */ -}} {{- $seen := .Page.Store.Get "ogpCardSeen" -}} {{- .Page.Store.Set "ogpCardSeen" true -}} {{- partial "helper/ogp-card.html" (dict "url" $dest "text" .Text "eager" (not $seen)) -}} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
