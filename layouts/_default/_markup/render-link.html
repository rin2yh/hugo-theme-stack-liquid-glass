{{- /* Link render hook. A bare external URL (an autolink, where the visible text equals the destination) is rendered as an Open Graph preview card. Every other link — labelled markdown links, relative/internal links, anchors, mailto — renders as a normal anchor, matching Hugo's default output. Set `params.ogpCard.enabled = false` in site config to fall back to plain links everywhere. */ -}} {{- $dest := .Destination -}} {{- $plain := .PlainText | default "" -}} {{- $external := or (hasPrefix $dest "http://") (hasPrefix $dest "https://") -}} {{- $isBareURL := and $external (eq (trim $plain " ") (trim $dest " ")) -}} {{- if $isBareURL -}} {{- /* Number the OGP cards in document order. Hugo renders (and memoizes) a page's content once, so the first card — ordinal 0 — is reliably the one nearest the top and the likely LCP element. It is fetched eagerly with a high priority; every later card stays lazy. */ -}} {{- $ordinal := .Page.Store.Get "ogpCardOrdinal" | default 0 -}} {{- .Page.Store.Set "ogpCardOrdinal" (add $ordinal 1) -}} {{- partial "helper/ogp-card.html" (dict "url" $dest "text" .Text "eager" (eq $ordinal 0)) -}} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
